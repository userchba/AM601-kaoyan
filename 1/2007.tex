已知 $f(x)$ 在 $(-\infty,+\infty)$ 内有二阶连续导数,且 $f(0)=0$, 又
\[ 
\varphi(x)=\left\{\begin{aligned}&f^{\prime}(0) && x=0 \\ 
	&\frac{e^{x}}{x} f(x) && x \neq 0\end{aligned}\right.
 \]
 求 $\varphi(x) ^{\prime} $。

\nynote{$\varphi(0)$处的导数要用定义另外求。}
\banswer{
	\textbf{$x\neq0$ 处.} 直接用商的求导法则：
	\begin{align*}
	\varphi^{\prime}(x)&=\left(\frac{e^{x}f(x)}{x}\right)^{\prime}\\
	&=\frac{xe^{x}f(x)+xe^{x}f^{\prime}(x)-e^{x}f(x)}{x^{2}} .
	\end{align*}

	\textbf{$x=0$ 处.} 这里 $\varphi$ 是分段定义的，必须按导数定义求。
	由 $f(0)=0$ 及带皮亚诺余项的泰勒公式
	\[
	f(x)=f^{\prime}(0)x+\frac{f^{\prime\prime}(0)}{2}x^{2}+o\left(x^{2}\right),\qquad
	e^{x}=1+x+o(x),
	\]
	相乘得
	\[
	e^{x}f(x)=f^{\prime}(0)x+\left(f^{\prime}(0)+\frac{f^{\prime\prime}(0)}{2}\right)x^{2}+o\left(x^{2}\right).
	\]
	于是
	\begin{align*}
	\varphi^{\prime}(0)&=\lim_{x\to0}\frac{\varphi(x)-\varphi(0)}{x}
	=\lim_{x\to0}\frac{e^{x}f(x)-xf^{\prime}(0)}{x^{2}}\\
	&=f^{\prime}(0)+\frac{f^{\prime\prime}(0)}{2} .
	\end{align*}

	综上
    \begin{equation*}
        \varphi^{\prime}(x)=
        \begin{cases}
            f^{\prime}(0)+\dfrac{f^{\prime\prime}(0)}{2},\qquad &x = 0,\\[8pt]
            \dfrac{x e^{x} f(x)+x e^{x} f^{\prime}(x)-e^{x} f(x)}{x^{2}},\qquad &x\neq 0 .\\
        \end{cases}
    \end{equation*}

	\par\vspace{1mm}\noindent
	\textbf{注：}{\kaishu
	原答案作 $\varphi^{\prime}(0)=f^{\prime}(0)+f^{\prime\prime}(0)$，有误：二阶项应带系数 $\frac12$。
	反例：取 $f(x)=x^{2}$，此时 $\varphi(x)\equiv xe^{x}$，$\varphi^{\prime}(0)=1$；
	而 $f^{\prime}(0)+f^{\prime\prime}(0)=2$，$f^{\prime}(0)+\frac12f^{\prime\prime}(0)=1$。

(本题满分 10 分)
求满足 $x=\int_{0}^{x} f(t) d t+\int_{0}^{x} t f(t-x) d t$ 的可微函数 $f(x)$。


\nynote{
    化简等式可以得到$f'(x) + f(-x) = 0$，两边求导，化简得$f''(x) + f(x) = 0$，
    解方程，套每步求导之前的等式可定$C_1$、$C_2$。
}
\banswer{
	\textbf{化简方程.} 在 $\int_{0}^{x}tf(t-x)dt$ 中令 $u=t-x$，则 $t=u+x$、$dt=du$，
	$u$ 由 $-x$ 变到 $0$，
	\[
	\int_{0}^{x}tf(t-x)dt=\int_{-x}^{0}(u+x)f(u)du .
	\]
	于是原方程成为
	\[
	x=\int_{0}^{x}f(t)dt+\int_{-x}^{0}uf(u)du+x\int_{-x}^{0}f(u)du .
	\]
	两端对 $x$ 求导（注意 $\frac{d}{dx}\int_{-x}^{0}g(u)du=g(-x)$）：
	\[
	1=f(x)-xf(-x)+\int_{-x}^{0}f(u)du+xf(-x),
	\]
	含 $xf(-x)$ 的两项相消，得
	\[
	f(x)+\int_{-x}^{0}f(u)du=1 .\qquad(\ast)
	\]

	\textbf{解出 $f$.} 在 $(\ast)$ 中令 $x=0$ 得 $f(0)=1$。对 $(\ast)$ 求导得
	\[
	f^{\prime}(x)+f(-x)=0 ,
	\]
	令 $x=0$ 得 $f^{\prime}(0)=-1$。再求导一次得 $f^{\prime\prime}(x)=f^{\prime}(-x)$；
	而在 $f^{\prime}(x)=-f(-x)$ 中以 $-x$ 代 $x$ 得 $f^{\prime}(-x)=-f(x)$，故
	\[
	f^{\prime\prime}(x)+f(x)=0 .
	\]
	其通解为 $f(x)=A\cos x+B\sin x$；由 $f(0)=1$、$f^{\prime}(0)=-1$ 得 $A=1$、$B=-1$，
	\[
	f(x)=\cos x-\sin x .
	\]
	（代回验证：$f^{\prime}(x)=-\sin x-\cos x=-f(-x)$，且 $f(0)=1$，$(\ast)$ 及原方程均成立。）
}

(本题满分 10 分)
若 $u=f(z y x), f(0)=0, f^{\prime}(1)=1$, 且 $\frac{\partial^{3} u}{\partial x \partial y \partial z}=x^{2} y^{2} z^{2} f^{\prime \prime \prime}(x y z)$, 求函数 $u$。

\nynote{
    展开偏导数，令$t=xyz$可得$f'(t) + 3tf''(t) = 0$，解方程。
}
\banswer{
	\textbf{展开三阶混合偏导.} 记 $t=xyz$，$u=f(t)$。逐次求偏导：
	\begin{align*}
	\frac{\partial u}{\partial z}&=xy\,f^{\prime}(t),\\
	\frac{\partial^{2}u}{\partial y\partial z}&=x^{2}yz\,f^{\prime\prime}(t)+x\,f^{\prime}(t),\\
	\frac{\partial^{3}u}{\partial x\partial y\partial z}
	&=x^{2}y^{2}z^{2}f^{\prime\prime\prime}(t)+3t\,f^{\prime\prime}(t)+f^{\prime}(t).
	\end{align*}
	与题设 $\frac{\partial^{3}u}{\partial x\partial y\partial z}=x^{2}y^{2}z^{2}f^{\prime\prime\prime}(xyz)$ 比较，得
	\[
	3t\,f^{\prime\prime}(t)+f^{\prime}(t)=0 .
	\]

	\textbf{解常微分方程.} 当 $t>0$ 时 $\frac{f^{\prime\prime}}{f^{\prime}}=-\frac{1}{3t}$，
	积分得 $\ln\abs{f^{\prime}}=-\frac13\ln t+\text{const}$，即 $f^{\prime}(t)=Ct^{-\frac13}$。
	由 $f^{\prime}(1)=1$ 得 $C=1$，故 $f^{\prime}(t)=t^{-\frac13}$，
	\[
	f(t)=\frac32t^{\frac23}+D .
	\]
	再由 $f(0)=0$ 得 $D=0$。所以
	\[
	u=f(xyz)=\frac32(xyz)^{\frac23}.
	\]
}

(本题满分 10 分)
设 $L$ 是分段光滑的简单闭曲线,且点 $(2,0),(-2,0)$ 均在闭曲线 $L$ 所围成区域的内部, 计算曲线积分 $I=\oint_{L}\left[\frac{y}{(2-x)^{2}+y^{2}}+\frac{y}{(2+x)^{2}+y^{2}}\right] d x+\left[\frac{2-x}{(2-x)^{2}+y^{2}}-\frac{2+x}{(2+x)^{2}+y^{2}}\right] d y$, 其中 $L$ 取正向。

\nynote{见西安交通大学出版社《高等数学典型题》\textbf{10-45}，积分里面是全微分，挖洞，斯托克斯公式。}

\banswer{
	\textbf{拆成两项.} 记
	\[
	\omega_{1}=\frac{y\,dx+(2-x)dy}{(2-x)^{2}+y^{2}},\qquad
	\omega_{2}=\frac{y\,dx-(2+x)dy}{(2+x)^{2}+y^{2}},
	\]
	则 $I=\oint_{L}\omega_{1}+\oint_{L}\omega_{2}$。直接验算可知：
	$\omega_{1}$ 在除 $(2,0)$ 外处处满足 $\frac{\partial Q}{\partial x}=\frac{\partial P}{\partial y}$，
	$\omega_{2}$ 在除 $(-2,0)$ 外亦然。

	\textbf{挖洞.} 取充分小的 $\varepsilon>0$，作正向小圆
	\[
	C_{1}:(x-2)^{2}+y^{2}=\varepsilon^{2},\qquad
	C_{2}:(x+2)^{2}+y^{2}=\varepsilon^{2},
	\]
	使它们互不相交且都含于 $L$ 内部。在 $L$ 与 $C_{1}\cup C_{2}$ 所夹的区域上
	被积表达式是全微分，由格林公式
	\[
	\oint_{L}=\oint_{C_{1}}+\oint_{C_{2}} .
	\]

	\textbf{逐个计算.} $\omega_{2}$ 的奇点 $(-2,0)$ 在 $C_{1}$ 之外，故 $\oint_{C_{1}}\omega_{2}=0$；
	同理 $\oint_{C_{2}}\omega_{1}=0$。在 $C_{1}$ 上分母恒为 $\varepsilon^{2}$，用格林公式
	\begin{align*}
	\oint_{C_{1}}\omega_{1}
	&=\frac{1}{\varepsilon^{2}}\oint_{C_{1}}y\,dx+(2-x)dy\\
	&=\frac{1}{\varepsilon^{2}}\iint_{D_{1}}(-1-1)\,d\sigma=-2\pi .
	\end{align*}
	同样地，在 $C_{2}$ 上
	\begin{align*}
	\oint_{C_{2}}\omega_{2}
	&=\frac{1}{\varepsilon^{2}}\oint_{C_{2}}y\,dx-(2+x)dy\\
	&=\frac{1}{\varepsilon^{2}}\iint_{D_{2}}(-1-1)\,d\sigma=-2\pi .
	\end{align*}
	故
	\[
	I=-2\pi-2\pi=-4\pi .
	\]

	\par\vspace{1mm}\noindent
	\textbf{注：}{\kaishu
	原题面 $dy$ 前第二项印作 $+\frac{2+x}{(2+x)^{2}+y^{2}}$。如此则该项不满足
	$\frac{\partial Q}{\partial x}=\frac{\partial P}{\partial y}$，积分值随 $L$ 而变（取
	$L$ 为椭圆、圆、其它闭曲线分别得 $-7.854$、$-6.283$、$-7.015$ 等），题目无确定答案；
	应为 $-\frac{2+x}{(2+x)^{2}+y^{2}}$，已据此改正。
	改正后对上述三条曲线数值积分均得 $-12.566371=-4\pi$，与原有答案一致。
	另外 $\omega_{1}=d\arctan\frac{y}{2-x}$、$\omega_{2}=-d\arctan\frac{y}{2+x}$，
	从辐角变化看也各贡献 $-2\pi$。
	}\par

(本题满分 10 分)
求方程 $4 x^{4} y^{\prime \prime \prime}-4 x^{3} y^{\prime \prime}+4 x^{2} y^{\prime}=1$ 的形如 $y=a x^{-1}$ 的特解，进而求该方程的通解。

\nynote{
    欧拉方程，
    令$x = e^t$可化为常系数方程，
    先解齐次，再带上特解就可以了。
}
\banswer{
	\textbf{求特解.} 设 $y=ax^{-1}$，则 $y^{\prime}=-ax^{-2}$、$y^{\prime\prime}=2ax^{-3}$、
	$y^{\prime\prime\prime}=-6ax^{-4}$，代入左端：
	\begin{align*}
	&4x^{4}\left(-6ax^{-4}\right)-4x^{3}\left(2ax^{-3}\right)+4x^{2}\left(-ax^{-2}\right)\\
	&\qquad=-24a-8a-4a=-36a .
	\end{align*}
	令 $-36a=1$ 得 $a=-\frac{1}{36}$，即特解 $y^{*}=-\frac{1}{36}x^{-1}$。

	\textbf{求齐次通解.} 齐次方程两端除以 $4x^{2}$（设 $x>0$）：
	\[
	x^{2}y^{\prime\prime\prime}-xy^{\prime\prime}+y^{\prime}=0 .
	\]
	令 $p=y^{\prime}$，得二阶欧拉方程 $x^{2}p^{\prime\prime}-xp^{\prime}+p=0$，
	其特征方程 $r(r-1)-r+1=(r-1)^{2}=0$ 有二重根 $r=1$，故
	\[
	p=\left(A+B\ln x\right)x .
	\]
	再积分（用 $\int x\ln x\,dx=\frac{x^{2}}{2}\ln x-\frac{x^{2}}{4}$）得
	\[
	y=\int p\,dx=C_{1}+\left(C_{2}+C_{3}\ln x\right)x^{2} .
	\]

	\textbf{通解.} 齐次通解加特解：
	\[
	y=C_{1}+\left(C_{2}+C_{3}\ln x\right)x^{2}-\frac{1}{36}x^{-1}.
	\]
	（代回原方程逐项验证，余项恰为 $1$。）
}

\item 
(本题满分 10 分)
在曲线 $\frac{x^{2}}{4}+y^{2}=1$ 上找到一个位于第一象限的点, 使得该曲线在该点处的切线与 该曲线以及 $x$ 轴和 $y$ 轴所围成的图形面积最小, 并求此最小面积。

\nynote{可设切线为$ \frac{x}{a} + \frac{y}{b} = 1$，$a$、$b$为截距，借$\Delta=0$利用基本不等式求解。}

\banswer{
	\textbf{写出面积.} 椭圆在第一象限的点可设为 $P\left(2\cos\theta,\sin\theta\right)$，
	$\theta\in\left(0,\frac\pi2\right)$，该点处的切线为
	\[
	\frac{x\cos\theta}{2}+y\sin\theta=1 ,
	\]
	它在两坐标轴上的截距分别为 $a=\frac{2}{\cos\theta}$、$b=\frac{1}{\sin\theta}$，
	与两轴围成的直角三角形面积
	\[
	S_{\triangle}=\frac{ab}{2}=\frac{1}{\sin\theta\cos\theta}=\frac{2}{\sin2\theta}.
	\]
	所求图形即该三角形挖去四分之一椭圆，而后者的面积
	$\frac{\pi\cdot2\cdot1}{4}=\frac{\pi}{2}$ 是常数，故
	\[
	S(\theta)=\frac{2}{\sin2\theta}-\frac{\pi}{2}.
	\]

	\textbf{求最小值.} $0<\sin2\theta\leqslant1$，等号当且仅当 $\theta=\frac\pi4$ 时成立，
	此时 $S$ 最小。故所求点与最小面积为
	\[
	P\left(\sqrt2,\ \frac{\sqrt2}{2}\right),\qquad
	S_{\min}=2-\frac{\pi}{2}\approx0.4292 .
	\]
	（数值极小化也给出 $\theta=0.785398=\frac\pi4$、$S=0.429204$。）
}



\item 
(本题满分 10 分)
当 $0<K=\sqrt{a^{2}+b^{2}}<r<R, \quad D: r^{2} \leqslant x^{2}+y^{2} \leqslant R^{2}$, 证明
\[
\frac{\pi\left(R^{2}-r^{2}\right)}{R+K} \leqslant \iint_{D} \frac{d \sigma}{\sqrt{(x-a)^{2}+(y-b)^{2}}} \leqslant \frac{\pi\left(R^{2}-r^{2}\right)}{r-K}
\]

\nynote{三角形三边关系不等式}

\banswer{
任取 $(x,y)\in D$，记 $\rho=\sqrt{x^2+y^2}$，由三角不等式得
$\rho-K\le\sqrt{(x-a)^2+(y-b)^2}\le\rho+K$。
由 $K<r\le\rho\le R$ 知 $\rho-K>0$，故
\[
\frac1{\rho+K}\le\frac1{\sqrt{(x-a)^2+(y-b)^2}}\le\frac1{\rho-K}.
\]
因此
\begin{align*}
\iint_D\frac{d\sigma}{\sqrt{(x-a)^2+(y-b)^2}}
&\ge\iint_D\frac1{\rho+K}\,d\sigma
\ge\iint_D\frac1{R+K}\,d\sigma
=\frac{\pi(R^2-r^2)}{R+K},\\[4pt]
\iint_D\frac{d\sigma}{\sqrt{(x-a)^2+(y-b)^2}}
&\le\iint_D\frac1{\rho-K}\,d\sigma
\le\iint_D\frac1{r-K}\,d\sigma
=\frac{\pi(R^2-r^2)}{r-K}.
\end{align*}
综上，所证不等式成立。
}


\item 
(本题满分 10 分)
设函数 $f(x)$ 在区间 $[0,1]$ 上可导, 且 $f(0)=0, f(1)=1$, 证明在区间 $[0,1]$ 上存在 两点 $x_{1}, x_{2}$, 使得 $\frac{1}{f^{\prime}\left(x_{1}\right)}+\frac{1}{f^{\prime}\left(x_{2}\right)}=2$。

\nynote{2009年十一题的一种特殊情况，令$f(x_0)= \frac{1}{2}$，再用拉格朗日中值定理就可以了。}

\banswer{
由连续函数的介值定理，存在 $x_0\in(0,1)$ 使得 $f(x_0)=\frac12$。
在 $[0,x_0]$ 和 $[x_0,1]$ 上分别应用拉格朗日中值定理，存在
$x_1\in(0,x_0)$，$x_2\in(x_0,1)$ 满足
\[
f'(x_1)=\frac{f(x_0)-f(0)}{x_0-0}=\frac{1}{2x_0},\qquad
f'(x_2)=\frac{f(1)-f(x_0)}{1-x_0}=\frac{1}{2(1-x_0)}.
\]
于是 $\displaystyle\frac1{f'(x_1)}+\frac1{f'(x_2)}=2x_0+2(1-x_0)=2$。
}


\item 
(本题满分 10 分)
设级数 $\sum\limits_{n=1}^{\infty} u_{n}$ 的各项 $u_{n}>0, n=1,2, \cdots,\left\{v_{n}\right\}$ 为一项正实数数列, 记 $a_{n}=\frac{u_{n} v_{n}}{u_{n+1}}-v_{n+1}$, 证明：如果 $\lim\limits _{n \rightarrow \infty} a_{n}=a$, 且 $a$ 为有限正数或者正无穷, 则 $\sum\limits_{n=1}^{\infty} u_{n}$ 收敛。


\banswer{
由 $a_n=\frac{u_n v_n}{u_{n+1}}-v_{n+1}$ 得 $u_n v_n-u_{n+1}v_{n+1}=a_n u_{n+1}$。
因 $\lim a_n=a>0$（或 $+\infty$），令
\[
c=\begin{cases}
a/2, & a\text{ 为有限正数},\\[2pt]
1,   & a=+\infty .
\end{cases}
\]
存在 $N$ 使 $n\ge N$ 时 $a_n\ge c>0$。
于是 $u_n v_n-u_{n+1}v_{n+1}=a_n u_{n+1}\ge c\,u_{n+1}>0$，
$\{u_n v_n\}_{n\ge N}$ 单调递减且有下界 $0$，故收敛。
对 $m>N$，
\[
u_N v_N-u_m v_m=\sum_{k=N}^{m-1}(u_k v_k-u_{k+1}v_{k+1})\ge c\sum_{k=N}^{m-1}u_{k+1}.
\]
令 $m\to\infty$，左端趋于有限值 $u_N v_N-\lim u_n v_n<\infty$，
故 $\sum_{k=N+1}^{\infty}u_k$ 收敛，从而 $\sum_{n=1}^{\infty}u_n$ 收敛。
}

	
	

\end{enumerate}
